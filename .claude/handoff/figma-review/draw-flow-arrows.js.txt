    const unnamed = edges.filter(e => e.root && !e.paired).map(e => lab(byId[e.from]) + ' ' + e.text + ' → ' + lab(byId[e.to]));
    const startName = MOB ? CONFIG.mstart + ', Device=Mobile' : CONFIG.start; const start = VS.find(c => c.name === startName || c.name === startName + ', Device=Desktop'); if (!start) throw new Error(WHICH + ' ' + U.dev + ' 기본 변형 없음');
    // 2. 가로 나무: 열 = 깊이, 행 = 부모 아래 차례
    function tree(roots, E) { const depth = {}, kids = {}, parentOf = {}, row = {}; const oi = k => { const i = ORDER.indexOf(box[k].label); return i < 0 ? 999 : i; }, pf = k => { const i = PFIRST.indexOf(box[k].label); return i < 0 ? 999 : i; };
      const order = []; for (const r of roots) { if (depth[r] !== undefined) continue; depth[r] = 0; order.push(r); let q = [r]; while (q.length) { const next = []; for (const n of q.slice().sort((a, b) => pf(a) - pf(b))) for (const e of E) if (e.a === n && depth[e.b] === undefined) { depth[e.b] = depth[n] + 1; parentOf[e.b] = n; (kids[n] = kids[n] || []).push(e.b); next.push(e.b); } q = next; } }
      for (const n in kids) kids[n] = kids[n].map((k, i) => [k, i]).sort((a, b) => oi(a[0]) - oi(b[0]) || a[1] - b[1]).map(x => x[0]);
      const placeRows = (n, r) => { row[n] = r; let h = 0; for (const k of (kids[n] || [])) h += placeRows(k, r + h); return Math.max(h, 1); }; let rows = 0; for (const r of order) rows += placeRows(r, rows);
      return { depth, kids, parentOf, row, rows, nodes: Object.keys(depth) }; }
    // 3. 그림 한 장: 층 L에 제목, 이름표, 글줄, 화살표를 그린다. 모습의 자리는 box[열쇠].x/y에 적는다
    async function diagram(L, title, roots, E, N, y0) {
      const T = tree(roots, E); const { depth, parentOf, row } = T;
      const fwdAll = E.filter(e => depth[e.a] !== undefined && depth[e.b] > depth[e.a]), back = E.filter(e => depth[e.a] !== undefined && !(depth[e.b] > depth[e.a]));
      const fwd = []; for (const e of fwdAll) { const same = fwd.find(x => x.a === e.a && x.b === e.b); if (same) { if (!same.text.split(' · ').includes(e.text)) same.text += ' · ' + e.text; } else fwd.push({ ...e }); }
      const far = []; for (let i = fwd.length - 1; i >= 0; i--) { const e = fwd[i]; if (parentOf[e.b] !== e.a && (EXTRA || Math.abs(row[e.b] - row[e.a]) > FAR || Math.min(box[e.a].h, box[e.b].h) < 200)) { far.push({ a: e.a, text: e.text + ' → ' + box[e.b].label }); fwd.splice(i, 1); } }
      for (const e of fwd) { e.t = await text(L, e.text, sBig, ink); e.t.name = 'do/' + e.text; }
      for (const k of T.nodes) { const B = box[k]; const t = await text(L, B.label, sBig, ink); t.name = 'name/' + B.label; B.nameT = t; B.nameId = t.id; B.nameW = t.width; }
      const maxD = Math.max(0, ...T.nodes.map(k => depth[k])); const colW = [], gap = [];
      for (let d = 0; d <= maxD; d++) { colW[d] = Math.max(...T.nodes.filter(k => depth[k] === d).map(k => MOB ? Math.max(box[k].w, box[k].nameW, (back.some(e => e.a === k) || N.some(n => n.a === k) || far.some(n => n.a === k)) ? 832 : 0) : box[k].w)); gap[d] = Math.max(MINGAP, STUB + 40 + Math.max(0, ...fwd.filter(e => depth[e.a] === d).map(e => e.t.width)) + 80); }
      const isExtra = e => parentOf[e.b] !== e.a;
      for (let d = 0; d < maxD; d++) { const n = fwd.filter(e => isExtra(e) && depth[e.b] - 1 === d).length; if (n) gap[d] += 120 + 40 * n; }
      const X = [xoff + PAD]; for (let d = 1; d <= maxD; d++) X[d] = X[d - 1] + colW[d - 1] + gap[d - 1];
      const tt = await text(L, title, sTitle, ink); tt.name = 'title/' + title; tt.x = xoff + PAD; tt.y = y0;
      let yy = y0 + tt.height + 160; const Y = [];
      for (let r = 0; r < T.rows; r++) { const inRow = T.nodes.filter(k => row[k] === r); const t = Math.max(0, ...inRow.map(k => box[k].t)), b = Math.max(0, ...inRow.map(k => box[k].b)), h = Math.max(...inRow.map(k => box[k].h)); Y[r] = yy + t + 96; yy += 96 + t + h + b + ROWGAP; }
      for (const k of T.nodes) { box[k].x = X[depth[k]]; box[k].y = Y[row[k]]; }
      for (const k of T.nodes) { const B = box[k];
        const t = B.nameT; t.x = B.x; t.y = B.y - B.t - t.height - 28;
        const groups = {}; for (const e of back.filter(e => e.a === k)) { const g = (groups[e.b] = groups[e.b] || []); if (!g.includes(e.text)) g.push(e.text); }
        const lines = Object.entries(groups).map(([to, ts]) => ts.join(' · ') + ' → ' + box[to].label).concat(far.filter(n => n.a === k).map(n => n.text)).concat(N.filter(n => n.a === k).map(n => n.text));
        if (lines.length) { const n = await text(L, lines.join('\n'), sSmall, muted, noteW(B)); n.name = 'back/' + B.label; n.x = B.x; n.y = B.y + B.h + B.b + 24; B.note = n.height + 24; B.noteId = n.id; } }
      for (let r = 0; r + 1 < T.rows; r++) { const inRow = T.nodes.filter(k => row[k] === r); const low = Math.max(...inRow.map(k => box[k].y + box[k].h + box[k].b + (box[k].note || 0))); const nextTop = Math.min(...T.nodes.filter(k => row[k] === r + 1).map(k => box[k].y - box[k].t)) - 200; if (low > nextTop) { const push = low - nextTop; const mine = new Set(T.nodes.filter(k => row[k] > r).flatMap(k => [box[k].nameId, box[k].noteId])); for (const k of T.nodes) if (row[k] > r) box[k].y += push; for (const c of L.children) if (mine.has(c.id)) c.y += push; } }
      const laneUsed = {};
      for (const e of fwd) { const s = box[e.a], d = box[e.b];
        const sx = s.x + s.w + 16, sy = s.y + s.h / 2, ex = s.x + (MOB ? colW[depth[e.a]] : s.w) + STUB, dx = d.x - 16, dy = d.y + d.h / 2; const extra = isExtra(e); let pts;
        if (!extra) pts = sy === dy ? [[sx, sy], [dx, dy]] : [[sx, sy], [ex, sy], [ex, dy], [dx, dy]];
        else { const k = laneUsed[depth[e.b]] = (laneUsed[depth[e.b]] || 0) + 1, lane = d.x - 100 - 40 * k, dir = Math.sign(dy - sy) || 1; const off = n => Math.round(Math.min(n.h / 2 - 8, Math.max(60, n.h / 4))); const sy2 = sy + dir * off(s), dy2 = sy === dy ? sy2 : dy - dir * off(d); pts = sy2 === dy2 ? [[sx, sy2], [dx, dy2]] : [[sx, sy2], [lane, sy2], [lane, dy2], [dx, dy2]]; }
        const v = figma.createVector(); L.appendChild(v); v.name = 'arrow/' + s.label + ' → ' + d.label;
        await v.setVectorNetworkAsync({ vertices: pts.map((p, i) => ({ x: p[0], y: p[1], strokeCap: i === pts.length - 1 ? 'ARROW_EQUILATERAL' : 'ROUND', strokeJoin: 'ROUND' })), segments: pts.slice(1).map((_, i) => ({ start: i, end: i + 1 })) });
        v.strokes = ink; v.strokeWeight = 8; v.fills = [];
        e.t.x = ex + 40; e.t.y = (extra ? pts[0][1] : dy) - e.t.height - 16; L.appendChild(e.t); e.pts = pts; e.extra = extra; }
      let right = X[maxD] + colW[maxD] + PAD, bottom = y0; for (const k of T.nodes) { bottom = Math.max(bottom, box[k].y + box[k].h + box[k].b + (box[k].note || 0)); right = Math.max(right, box[k].x + Math.max(box[k].note ? noteW(box[k]) : 0, box[k].nameW || 0) + PAD); }
      const D = { title, fwd, back, far, nodes: T.nodes, notes: N.filter(n => depth[n.a] !== undefined), L }; drawn.push(D);
      return { right, bottom: bottom + 200, nodes: T.nodes, depth, D };
    }
    // 4. 흐름 그림(손·마우스): 키가 아닌 흐름으로 기본에서 닿는 모습. 진짜 변형을 여기에 놓는다
    const allE = edges.map(e => ({ a: K(e.from), b: K(e.to), text: e.text })); const reachAll = new Set(tree([K(start.id)], allE).nodes);
    const orphan = VS.filter(c => !reachAll.has(K(c.id))).map(c => K(c.id));
    const inMouse = new Set(tree([K(start.id), ...orphan], edges.filter(e => !e.key).map(e => ({ a: K(e.from), b: K(e.to), text: e.text }))).nodes);
    for (const e of edges) if (!e.key && !inMouse.has(K(e.from))) e.key = true; for (const n of notes) if (!n.key && !inMouse.has(K(n.from))) n.key = true;
    const m = await diagram(layer, U.titles[0], [K(start.id), ...orphan], edges.filter(e => !e.key).map(e => ({ a: K(e.from), b: K(e.to), text: e.text })), notes.filter(n => !n.key).map(n => ({ a: K(n.from), text: n.text })), 120);
    // 5. 키보드 흐름 그림: 같은 기본에서 시작해 키 줄로 닿는 모습. 위 그림에 놓인 모습은 그 변형의 인스턴스('shot/…'), 키로만 닿는 모습은 진짜 변형
    for (const c of VS) box[Q(c.id)] = { ...box[K(c.id)], x: undefined, y: undefined, note: 0, noteId: undefined, nameId: undefined, inst: inMouse.has(K(c.id)) };
    const keyE = edges.filter(e => e.key).map(e => ({ a: Q(e.from), b: Q(e.to), text: e.text, line: e.line })), keyN = notes.filter(n => n.key).map(n => ({ a: Q(n.from), text: n.text, line: n.line }));
    const kroots = [Q(start.id), ...orphan.map(o => Q(o.slice(2))).filter(q => keyE.some(e => e.a === q) || keyN.some(n => n.a === q))];
    const k2 = await diagram(klayer, U.titles[1], kroots, keyE, keyN, m.bottom + 200);
    const unreachedLines = keyE.concat(keyN).filter(e => k2.depth[e.a] === undefined).map(e => e.line || (box[e.a].label + ' · ' + e.text));
    for (const k of k2.nodes) { const B = box[k]; if (!B.inst) continue; const i = byId[B.id].createInstance(); klayer.insertChild(0, i); i.name = 'shot/' + B.label; i.x = B.x; i.y = B.y; }
    for (const c of VS) { const B = inMouse.has(K(c.id)) ? box[K(c.id)] : box[Q(c.id)]; if (!B || B.x === undefined) throw new Error(WHICH + ' ' + U.dev + ' 놓이지 않은 모습: ' + c.name); }
    if (CONFIG.backdrop && U.bds.length) bdJobs.push([U.bds[0], m.nodes], [U.bds[1], k2.nodes]);
    const kset = new Set(k2.nodes.map(k => k.slice(2)));
    per[U.dev] = { variants: VS.length, unnamed, dupLabels, crossDropped, orphan: orphan.map(k => box[k].label), keyboardTop: m.bottom + 200, colLeft: xoff, colRight: Math.max(m.right, k2.right),
      diagrams: [m.D, k2.D].map(d => ({ title: d.title, nodes: d.nodes.length, arrows: d.fwd.length, backNotes: d.back.length, farNotes: d.far.length, notes: d.notes.length })),
      keyNamesInTouch: m.D.fwd.concat(m.D.back).map(e => e.text).concat(m.D.notes.map(n => n.text.split(' → ')[0])).filter(t => t.split(' · ').some(isKeyTok)),
      notInKeyboard: m.nodes.filter(k => !kset.has(k.slice(2))).map(k => box[k].label), keyOnly: k2.nodes.filter(k => !box[k].inst).map(k => box[k].label), unreachedLines };
    xoff = Math.max(m.right, k2.right) + COLGAP; bottomAll = Math.max(bottomAll, k2.bottom);
  }
  // 6. 세트와 층의 크기, 변형의 자리, 바탕, 섹션
  set.resizeWithoutConstraints(xoff - COLGAP, bottomAll + 60);
  for (const c of set.children) { const B = box[K(c.id)].x !== undefined ? box[K(c.id)] : box[Q(c.id)]; c.x = B.x; c.y = B.y; }
  host.resize(set.x + set.width, set.y + set.height); for (const l of host.children.filter(c => ARROWS.includes(c.name))) { l.x = set.x; l.y = set.y; l.resize(set.width, set.height); }
  if (bdJobs.length) { const wp = await figma.getNodeByIdAsync(CONFIG.backdrop);
    for (const [nm, nodes] of bdJobs) { const bd = figma.createFrame(); bd.name = nm; bd.fills = []; bd.clipsContent = false; host.insertChild(bdAt++, bd); bd.x = set.x; bd.y = set.y; bd.resize(set.width, set.height);
      for (const k of nodes) { const B = box[k]; const t = B.t + 24; const f = figma.createFrame(); f.name = 'backdrop/' + B.label; f.fills = []; f.clipsContent = true; bd.appendChild(f); f.x = B.x - 24; f.y = B.y - t; f.resize(B.w + 48, B.h + t + B.b + 24); const w = wp.createInstance(); f.appendChild(w); w.rescale(Math.max(f.width / w.width, f.height / w.height)); w.x = Math.round((f.width - w.width) / 2); w.y = f.height - w.height; } } }
  if (sectionNode) { const delta = rootFrame.y + rootFrame.height - rootBottomBefore; for (const c of sectionNode.children) if (c !== rootFrame && c.y >= rootBottomBefore - 1) c.y += delta;
    let right = 0, bottom = 0; for (const c of sectionNode.children) { right = Math.max(right, c.x + c.width); bottom = Math.max(bottom, c.y + c.height); } sectionNode.resizeWithoutConstraints(right + 80, bottom + 80);
    const grow = sectionNode.y + sectionNode.height - secBottomBefore; if (grow !== 0) for (const o of page.children) if (o.type === 'SECTION' && o !== sectionNode && o.y >= secBottomBefore - 1 && o.x < sectionNode.x + sectionNode.width && o.x + o.width > sectionNode.x) o.y += grow; }
  // 7. 검사: 어긋난 것을 bad에 모아 돌려준다
  const bad = []; const sa = set.absoluteBoundingBox;
  if (host.layoutMode !== 'NONE') bad.push('세트를 담은 프레임이 자동 배치다');
  for (const l of host.children.filter(c => LAYERS.includes(c.name))) { if (l.x !== set.x || l.y !== set.y || l.width !== set.width || l.height !== set.height) bad.push(l.name + '의 자리·크기가 세트와 다르다'); const out = l.children.find(c => { const b = c.absoluteBoundingBox; return b.x < sa.x - 1 || b.y < sa.y - 1 || b.x + b.width > sa.x + sa.width + 1 || b.y + b.height > sa.y + sa.height + 1; }); if (out) bad.push(l.name + ' 안의 ' + out.name + '가 세트 밖에 있다'); }
  for (const stray of page.findAll(n => LAYERS.includes(n.name) && n.parent && !n.parent.children.some(c => c.type === 'COMPONENT_SET'))) bad.push(stray.name + ' ' + stray.id + '가 세트 옆이 아닌 곳에 있다');
  if (sectionNode) { const ha = host.absoluteBoundingBox, ca = { x: sectionNode.absoluteTransform[0][2], y: sectionNode.absoluteTransform[1][2] }; if (ha.x < ca.x || ha.y < ca.y || ha.x + ha.width > ca.x + sectionNode.width || ha.y + ha.height > ca.y + sectionNode.height) bad.push('흐름도가 섹션 밖으로 나갔다');
    for (const c of sectionNode.children) if (c !== rootFrame && c.y < rootFrame.y + rootFrame.height && c.y + c.height > rootFrame.y && c.x < rootFrame.x + rootFrame.width && c.x + c.width > rootFrame.x) bad.push('섹션의 ' + c.name + '가 흐름도와 겹친다'); }
  const hit = (a, b) => a.x < b.x + b.w && a.x + a.w > b.x && a.y < b.y + b.h && a.y + a.h > b.y;
  const TX = host.children.filter(c => ARROWS.includes(c.name)).flatMap(l => l.children.filter(c => c.type === 'TEXT').map(t => ({ n: t.name, x: t.x, y: t.y, w: t.width, h: t.height, id: t.id })));

  for (const t of TX) for (const v of VB) if (hit(t, v)) bad.push('글자가 모습과 겹친다: ' + t.n + ' / ' + v.n);
  for (let i = 0; i < VB.length; i++) for (let k = i + 1; k < VB.length; k++) if (hit(VB[i], VB[k])) bad.push('모습이 겹친다: ' + VB[i].n + ' / ' + VB[k].n);
  let crossings = 0; const allFwd = drawn.flatMap(d => d.fwd);
  const segs = []; for (const e of allFwd) for (let i = 0; i + 1 < e.pts.length; i++) { const a = e.pts[i], b = e.pts[i + 1]; segs.push({ e, h: a[1] === b[1], c: a[1] === b[1] ? a[1] : a[0], lo: a[1] === b[1] ? Math.min(a[0], b[0]) : Math.min(a[1], b[1]), hi: a[1] === b[1] ? Math.max(a[0], b[0]) : Math.max(a[1], b[1]) }); }
  const nm = e => box[e.a].label + ' → ' + box[e.b].label;
  for (const g of segs) { const r = g.h ? { x: g.lo - 4, y: g.c - 4, w: g.hi - g.lo + 8, h: 8 } : { x: g.c - 4, y: g.lo - 4, w: 8, h: g.hi - g.lo + 8 }; for (const v of VB) if (v.k !== g.e.a && v.k !== g.e.b && hit(r, v)) bad.push('화살표가 모습을 지난다: ' + nm(g.e) + ' / ' + v.n); for (const t of TX) if (t.id !== g.e.t.id && hit(r, t)) bad.push('화살표가 글자를 지난다: ' + nm(g.e) + ' / ' + t.n); }
  for (let i = 0; i < segs.length; i++) for (let k = i + 1; k < segs.length; k++) { const A = segs[i], B = segs[k]; if (A.e === B.e) continue; const family = A.e.a === B.e.a && !A.e.extra && !B.e.extra;
    if (A.h === B.h) { if (!family && Math.abs(A.c - B.c) < 12 && Math.min(A.hi, B.hi) - Math.max(A.lo, B.lo) > 0) bad.push('선이 같은 구간을 함께 쓴다: ' + nm(A.e) + ' / ' + nm(B.e)); }
    else { const Hs = A.h ? A : B, Vv = A.h ? B : A; if (Vv.c > Hs.lo + 1 && Vv.c < Hs.hi - 1 && Hs.c > Vv.lo + 1 && Hs.c < Vv.hi - 1) crossings++; } }
  for (const e of allFwd) { const tip = e.pts[e.pts.length - 1]; for (const g of segs) { if (g.e === e) continue; const nearTip = g.h ? (Math.abs(g.c - tip[1]) < 60 && g.lo < tip[0] + 60 && g.hi > tip[0] - 60) : (Math.abs(g.c - tip[0]) < 60 && g.lo < tip[1] + 60 && g.hi > tip[1] - 60); if (nearTip && !(g.e.a === e.a && !g.e.extra && !e.extra)) bad.push('선이 다른 화살촉 근처를 지난다: ' + nm(g.e) + ' / ' + nm(e)); } }
  for (const a of allFwd) for (const b of allFwd) if (a !== b && a.t.x < b.t.x + b.t.width && a.t.x + a.t.width > b.t.x && b.t.y - (a.t.y + a.t.height) >= 0 && b.t.y - (a.t.y + a.t.height) < 60) bad.push('한 가로 선에 글자가 둘 붙는다: ' + a.t.name + ' / ' + b.t.name);
  const secs = page.children.filter(n => n.type === 'SECTION');
  for (let p = 0; p < secs.length; p++) for (let q2 = p + 1; q2 < secs.length; q2++) { const A = secs[p], B = secs[q2]; if (A.x < B.x + B.width && A.x + A.width > B.x && A.y < B.y + B.height && A.y + A.height > B.y) bad.push('섹션이 겹친다: ' + A.name + ' / ' + B.name); }
  // 8. 결과
  let rx = 0; for (const c of set.children) for (const x of [c, ...c.findAll(z => { try { return z.reactions && z.reactions.length > 0; } catch (e) { return false; } })]) for (const r of x.reactions) rx += r.actions.filter(a => a.type === 'NODE').length;
  results[WHICH] = { bad: bad.slice(0, 14), badCount: bad.length, crossings, variants: set.children.length, reactionsInSet: rx, setXY: [set.x, set.y, set.width, set.height].join(','), section: sectionNode ? [sectionNode.width, sectionNode.height].join('x') : null, sectionKids: sectionNode ? sectionNode.children.map(c => c.name + ' ' + Math.round(c.x) + ',' + Math.round(c.y) + ' ' + Math.round(c.width) + 'x' + Math.round(c.height)) : null, per };
} catch (e) { results[WHICH] = { error: String(e && e.message || e) }; } }
return results;
